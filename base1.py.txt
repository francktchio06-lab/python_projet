def create_table():
    conn = sqlite3.connect('base.db')
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS enseignant(
          
            Nom TEXT NOT NULL\,
            Prenoms TEXT NOT NULL,
            heure_jour INT NOT NULL,
            heure_total INT NOT NULL,
            Filiere TEXT NOT NULL, 
            Adresse TEXT NOT NULL)''')
    conn.commit()
    conn.close()
